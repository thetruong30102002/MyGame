export interface ArchiveProgress { level: number; evidence: readonly string[]; read: readonly string[] }
const levelOf=(n:number)=>Number.isFinite(n)?Math.max(0,Math.floor(n)):0;
/** Luật thiết kế thuần để thử nội dung; chưa nối vào save hoặc mở khóa map sản xuất. */
export function archiveAccess(progress: ArchiveProgress): string | null {
	if(levelOf(progress.level)<SKY_CASTLE_DESIGN.access.minLevel)return 'Đề xuất: cần cấp '+SKY_CASTLE_DESIGN.access.minLevel+' để lên Thiên Thành.';
	const missing=SKY_CASTLE_DESIGN.access.evidence.filter(id=>!progress.evidence.includes(id));
	return missing.length?'Cần đối chiếu: '+missing.map(id=>ARCHIVE_EVIDENCE.find(e=>e.id===id)?.name||id).join(', '):null;
}
export function archiveChapterGate(id:string,progress:ArchiveProgress):string|null {
	const chapter=ARCHIVE_CHAPTERS.find(c=>c.id===id);
	if(!chapter)return 'Không có chương thư khố này.';
	const access=archiveAccess(progress);if(access)return access;
	// Không tin cờ đọc chương sau nếu các chương trước chưa thực sự đủ điều kiện.
	if(chapter.previous){const previous=archiveChapterGate(chapter.previous,progress);if(previous)return previous;if(!progress.read.includes(chapter.previous))return 'Đọc chương trước: '+ARCHIVE_CHAPTERS.find(c=>c.id===chapter.previous)!.title;}
	const missing=chapter.requires.filter(e=>!progress.evidence.includes(e));
	return missing.length?'Thiếu chứng cứ: '+missing.map(id=>ARCHIVE_EVIDENCE.find(e=>e.id===id)?.name||id).join(', '):null;
}
/** Giao diện người chơi tương lai chỉ nhận nội dung chương đã mở, không lộ đoạn bị khóa. */
export function visibleArchive(progress:ArchiveProgress):ArchiveChapter[]{return ARCHIVE_CHAPTERS.filter(c=>!archiveChapterGate(c.id,progress));}
