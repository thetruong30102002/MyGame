function weaponAt(x:number,y:number,s:number[],c:readonly number[]){if(!s.length)return false;const[a,b,d,e]=s,dx=d!-a!,dy=e!-b!,len=Math.hypot(dx,dy),t=((x-a!)*dx+(y-b!)*dy)/(len*len),perp=Math.abs((x-a!)*dy-(y-b!)*dx)/len;if(t>=0&&t<=1&&perp<=14)return true;return Math.abs(t*len)<12&&perp<38&&c[0]!>165&&c[1]!>100&&c[2]!<115;}
function hairTop(im:RGBAImage,cx:number,slot:number,seg:number[]){for(let y=305;y<455;y++){let n=0;for(let x=Math.round(cx-96);x<=cx+96;x++){if(Math.abs(x-cx)<22||x<slot*im.width/4||x>=(slot+1)*im.width/4)continue;const c=getPixel(im,x,y);if(c[3]<235||weaponAt(x,y,seg,c))continue;const[r,g,b]=c;if(r>45&&r<235&&Math.abs(r-g)<30&&b>=g-3&&b>=r-8&&b<r+60)n++;}if(n>=10)return y;}throw Error('Không xác định được đỉnh tóc ô'+slot);}
function flipLegs(im:RGBAImage){const out={...im,data:new Uint8Array(im.data)},seen=new Set<number>(),q:number[][]=[];for(let x=130;x<=158;x++)if(getPixel(im,x,201)[3])q.push([x,201]);while(q.length){const[x,y]=q.pop()!,k=y!*288+x!;if(seen.has(k)||y!<201||y!>260||x!<112||x!>176)continue;const c=getPixel(im,x!,y!);if(!c[3]||c[2]>c[0]+14||c[1]>c[0]+8)continue;seen.add(k);for(const[dx,dy]of [[1,0],[-1,0],[0,1],[0,-1]])q.push([x!+dx!,y!+dy!]);}for(const k of seen)setPixel(out,k%288,Math.floor(k/288),[0,0,0,0]);for(const k of seen)setPixel(out,288-k%288,Math.floor(k/288),getPixel(im,k%288,Math.floor(k/288)));return out;}

function recoverMaskHoles(body:RGBAImage,front:RGBAImage,full:RGBAImage,bottom:number){const seen=new Set<number>();for(let y=75;y<bottom;y++)for(let x=112;x<176;x++){const key=y*288+x;if(seen.has(key)||getPixel(body,x,y)[3])continue;const q=[[x,y]],pts:number[][]=[];let boundary=false;seen.add(key);while(q.length){const[a,b]=q.pop()!;pts.push([a!,b!]);for(const[dx,dy]of [[1,0],[-1,0],[0,1],[0,-1]]){const X=a!+dx!,Y=b!+dy!,k=Y*288+X;if(X<112||X>=176||Y<75||Y>=bottom){boundary=true;continue;}if(seen.has(k)||getPixel(body,X,Y)[3])continue;seen.add(k);q.push([X,Y]);}}if(!boundary&&pts.length<=6&&pts.every(([X,Y])=>getPixel(full,X!,Y!)[3]))for(const[X,Y]of pts){const c=getPixel(full,X!,Y!);setPixel(body,X!,Y!,c);setPixel(front,X!,Y!,c);}}}
